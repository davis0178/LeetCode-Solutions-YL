```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        @cache
        def dfs(i: int, c: int) -> int:
            if i < 0:
                return 0 if c == 0 else inf # 凑不齐返回 inf
            if c < coins[i]: # 只能不选
                return dfs(i - 1, c)
            # 不选 vs 继续选，取最小
            return min(dfs(i - 1, c), dfs(i, c - coins[i]) + 1)
        
        # 入口，从大的开始
        ans = dfs(len(coins) - 1, amount)
        return ans if ans < inf else -1
```

记得有一个只能不选的边界条件，那就是当前种类硬币 `i` 的面值大于剩余金额 `c`

```python
class Solution:
    def coinChange(self, coins: List[int], amount: int) -> int:
        n = len(coins)
        
        # 行：第 i 中硬币。会小于 0，所以整体延后一格
        # 列：表示当前要凑出的目标金额
        # 值：凑出该金额所需的最少硬币数量
        f = [[inf] * (amount + 1) for _ in range(n + 1)] 
        f[0][0] = 0 # 没有硬币可用，且目标金额为 0，所需硬币也为 0（递推入口）

        for i, x in enumerate(coins): # 遍历每一种硬币
            for c in range(amount + 1): # 遍历每一种金额（背包容量）
                if c < x: # 进不去，继承上一行的结果
                    f[i + 1][c] = f[i][c]
                else: # 进得去，在不放与放之间取最小值
                    f[i + 1][c] = min(f[i][c], f[i + 1][c - x] + 1)
        ans = f[n][amount]
        return ans if ans < inf else -1
```
