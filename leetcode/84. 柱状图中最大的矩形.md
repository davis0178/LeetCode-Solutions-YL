```python
class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
	    n = len(heights)
	    left = [-1] * n # 不存在左更小高度，则为 -1
	    st = []
	    for i, h in enumerate(heights): # 找左边，正向遍历
		    while st and heights[st[-1]] >= h: # 栈顶是最近的左高度
			    st.pop()
			if st:
				left[i] = st[-1]
			st.append(i)
		
		right = [n] * n # 不存在右最小高度，则为 n
		st.clear()
		for i in range(n - 1, -1, -1): # 找右边，反向遍历
			h = heights[i]
			while st and heights[st[-1]] >= h: # 栈顶是最近的右高度
				st.pop()
			if st:
				right[i] = st[-1]
			st.append(i)
		
		ans = 0
		for h, l, r in zip(heights, left, right):
			ans = max(ans, h * (r - l - 1))
		return ans
```

遍历每个 `h = heights[i]`，找左侧小于 `h` 的最近元素下标 `left` 和右侧小于 `h` 的最近元素下标 `right`，那么 `left + 1` 和 `right - 1` 就是矩形最左边和最右边的那根柱子，这样就能对每个高度最大化矩形宽度，遍历到最大的答案

如何快速找到 `left` 和 `right`？其实就是做两次**每日温度**（**单调栈**）
